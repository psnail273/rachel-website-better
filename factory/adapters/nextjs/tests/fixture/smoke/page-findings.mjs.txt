// A stand-in for the smoke step's browser script (_smoke_page.mjs), for tests
// that need its findings without a browser: it prints the file STAND_IN_FINDINGS
// names, and exits 1 when that holds any.
import { readFileSync } from "node:fs";

const findings = readFileSync(process.env.STAND_IN_FINDINGS, "utf8");
process.stdout.write(findings);
process.exit(findings === "" ? 0 : 1);
