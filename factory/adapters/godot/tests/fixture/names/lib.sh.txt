# A Godot tests/lib.sh for names_test.sh: a helper that skips without each real
# tool (Godot, the formatter, gdmutant), and one that doesn't.

lib_godot() {
	[ -n "${GODOT_BIN:-}" ] || skip "GODOT_BIN is not set"
}

lib_formatter() {
	[ "$(gdscript-formatter --version)" = "gdscript-formatter 0.27.0" ] ||
		skip "no gdscript-formatter 0.27.0 (set GDSCRIPT_FORMATTER, or put it on PATH)"
}

lib_gdmutant() {
	[ "$(gdmutant --version)" = "gdmutant 0.1.4" ] || skip 'no gdmutant 0.1.4 (set GDMUTANT_BIN, or put it on PATH)'
}

lib_plain() {
	echo plain
}

# A helper the check can't read, at the end of the file: no closing brace at
# column 0.
lib_unread() {
	echo unread
	}
