func test_lambda_guard_in_a_loop() -> void:
	var callbacks: Array[Callable] = []
	for item: int in [1, 2]:
		var callback := func(x: int) -> int:
			if x < 0: return -1
			return x + item
		callbacks.append(callback)
	assert_int(callbacks.size()).is_equal(2)
