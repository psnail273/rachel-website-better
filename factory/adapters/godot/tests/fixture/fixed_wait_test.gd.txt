extends GdUnitTestSuite


func test_waits() -> void:
	await await_millis(200)
	await get_tree().create_timer(0.5).timeout
	OS.delay_msec(100)
	var guard: SceneTreeTimer = get_tree().create_timer(3.0, true, false, true)
	while guard.time_left > 0.0:
		await get_tree().process_frame
	# fixed-wait: no Serve may come before the delay
	await await_millis(200)
	await await_millis(100) # fixed-wait: the overlay stays shut this long
	var text: String = "await await_millis(5)"
	assert_str(text).is_not_empty()
