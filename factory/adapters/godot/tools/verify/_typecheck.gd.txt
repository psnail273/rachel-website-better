extends SceneTree
## The typecheck step's loader. tools/verify/typecheck copies it out of the
## project as a .gd file and runs it with:
##   godot --headless --path . -s <this> -- <file list> <warnings left off>
## <file list> holds one project-relative .gd path per line; <warnings left off>
## is a comma-separated list of warning names that may stay below Error.
## It prints, on stdout:
##   TYPECHECK-LEVEL <warning> <level>   a warning setting below Error (2)
##   TYPECHECK-RULE <directory>          a directory_rules entry that excludes a folder
##   TYPECHECK-FAIL <path>               a script that didn't load or can't instantiate
## Godot prints each script's parse errors on stderr itself.

const PREFIX := "debug/gdscript/warnings/"
const ERROR_LEVEL := 2


func _init() -> void:
	var args := OS.get_cmdline_user_args()
	if args.size() != 2:
		printerr("usage: -- <file list> <warnings left off>")
		quit(2)
		return
	var off := args[1].split(",", false)
	_check_levels(off)
	_check_directory_rules()
	quit(0 if _check_scripts(args[0]) else 2)


func _check_levels(off: PackedStringArray) -> void:
	for property: Dictionary in ProjectSettings.get_property_list():
		var name_: String = property["name"]
		var type_: int = property["type"]
		if not name_.begins_with(PREFIX) or type_ != TYPE_INT:
			continue
		var warning := name_.trim_prefix(PREFIX)
		if warning in off:
			continue
		var level: int = ProjectSettings.get_setting(name_)
		if level != ERROR_LEVEL:
			print("TYPECHECK-LEVEL %s %d" % [warning, level])


func _check_directory_rules() -> void:
	var rules: Dictionary = ProjectSettings.get_setting(PREFIX + "directory_rules", {})
	for directory: Variant in rules:
		var rule: int = rules[directory]
		if rule == 0 and str(directory) != "res://addons":
			print("TYPECHECK-RULE %s" % str(directory))


func _check_scripts(list_path: String) -> bool:
	var list := FileAccess.open(list_path, FileAccess.READ)
	if list == null:
		printerr("can't read the file list '%s'" % list_path)
		return false
	while not list.eof_reached():
		var path := list.get_line().strip_edges()
		if path.is_empty():
			continue
		var script := ResourceLoader.load("res://" + path, "", ResourceLoader.CACHE_MODE_IGNORE) as Script
		if script == null or not script.can_instantiate():
			print("TYPECHECK-FAIL %s" % path)
	return true
